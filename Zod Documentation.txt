    const canvas = document.getElementById("canvas");
    const ctx = canvas.getContext("2d");

    const state = {
      1: true, // Input Pool (&r == xora#FFFFFF)
      2: true, // Compute Layer (platonic_shapes(1, 1))

      const badge = document.getElementById(`status-${id}`);
      card.classList.toggle("active", state[id]);
      badge.textContent = state[id] ? (id === 1 ? "READY" : "ACTIVE") : "IDLE";
    }

    function resize() {
      canvas.width = window.innerWidth - 320;
      canvas.height = window.innerHeight;
    }
    window.addEventListener("resize", resize);
    resize();

    // Base Platonic icosahedron-inspired vertices
    const phi = (1 + Math.sqrt(5)) / 2;
    const baseVertices = [
      [-1,  phi, 0], [ 1,  phi, 0], [-1, -phi, 0], [ 1, -phi, 0],
      [ 0, -1,  phi], [ 0,  1,  phi], [ 0, -1, -phi], [ 0,  1, -phi],
      [ phi, 0, -1], [ phi, 0,  1], [-phi, 0, -1], [-phi, 0,  1]
    ];

    const edges = [
      [0,1],[0,5],[0,7],[0,10],[0,11],[1,5],[1,7],[1,8],[1,9],
      [2,3],[2,4],[2,6],[2,10],[2,11],[3,4],[3,6],[3,8],[3,9],
      [4,5],[4,9],[4,11],[5,9],[5,11],[6,7],[6,8],[6,10],[7,8],[7,10],
      [8,9],[10,11]
    ];

    // Stage 01: Bitwise XOR against 0xFFFFFF reference
    function xorFilter(val) {
      return state[1] ? ((val ^ 0xFFFFFF) & 0xFF) / 255 : 0.5;
    }

    // Stage 02: 3D rotation & Inverse Metaverse perspective projection
    function project(v, angle, scale, warp) {
      let [x, y, z] = v;

      // Rotate Y and X
      const cosY = Math.cos(angle), sinY = Math.sin(angle);
      const x1 = x * cosY - z * sinY;
      const z1 = z * cosY + x * sinY;

      const cosX = Math.cos(angle * 0.6), sinX = Math.sin(angle * 0.6);
      const y2 = y * cosX - z1 * sinX;
      const z2 = z1 * cosX + y * sinX;

      // Inverse perspective: objects further away expand outward
      const invZ = 1 / Math.max(0.25, 3.2 - z2 * warp);
      return {
        x: canvas.width / 2 + x1 * scale * invZ,
        y: canvas.height / 2 + y2 * scale * invZ,
        depth: z2
      };
    }

    let tick = 0;

    function render() {
      tick += 0.015;
      const warp = parseFloat(document.getElementById("warp").value);
      const density = parseInt(document.getElementById("density").value, 10);
      document.getElementById("warp-val").textContent = warp.toFixed(1);
      document.getElementById("density-val").textContent = density;

      // Stage 04: Execution Layer background clear / trail
      ctx.fillStyle = state[4] ? "rgba(5, 5, 8, 0.25)" : "#050508";
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      if (!state[4]) {
        requestAnimationFrame(render);
        return;
      }

      const seedOffset = xorFilter(Math.floor(tick * 60));

      // Render nested inverse vector shells
      for (let layer = 1; layer <= density; layer++) {
        const ratio = layer / density;
        const scale = 90 + ratio * 220;
        const angle = tick + ratio * Math.PI * (state[1] ? seedOffset + 0.5 : 0.2);

        const projected = baseVertices.map(v => project(v, angle, scale, warp));

        // Stage 03: Linking / Reasoning via PI-mapped spectral hue
        const hue = state[3]
          ? ((layer * Math.PI * 36) + tick * 60) % 360
          : 200;

        ctx.strokeStyle = `hsla(${hue}, 90%, 60%, ${0.15 + (1 - ratio) * 0.55})`;
        ctx.lineWidth = state[2] ? 1.4 : 0.6;

        if (state[2]) {
          // Draw Platonic vector wireframe
          ctx.beginPath();
          for (const [a, b] of edges) {
            ctx.moveTo(projected[a].x, projected[a].y);
            ctx.lineTo(projected[b].x, projected[b].y);
          }
          ctx.stroke();
        } else {
          // Fallback raw point cloud when Compute Layer is Idle
          for (const p of projected) {
            ctx.fillStyle = `hsla(${hue}, 90%, 65%, 0.8)`;
            ctx.fillRect(p.x - 1.5, p.y - 1.5, 3, 3);
          }
        }
      }

      requestAnimationFrame(render);
    }

    render();
  </script>

import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { z } from "zod";

// Pipeline state matching your 4-stage control structure
const pipelineState = {
  inputPool: { status: "READY", expression: "&r == xora#FFFFFF", seed: 0xffffff },
  computeLayer: { status: "IDLE", model: "platonic_shapes(1, 1)", warp: 1.0, density: 16 },
  linkingReasoning: { status: "IDLE", mapping: "rainbow_tables(PI)", phase: Math.PI },
  executionLayer: { status: "IDLE", output: "pixels(r)", lastVectorCount: 0 }
};

// Base Platonic icosahedron vertices
const PHI = (1 + Math.sqrt(5)) / 2;
const BASE_VERTICES = [
  [-1,  PHI, 0], [ 1,  PHI, 0], [-1, -PHI, 0], [ 1, -PHI, 0],
  [ 0, -1,  PHI], [ 0,  1,  PHI], [ 0, -1, -PHI], [ 0,  1, -PHI],
  [ PHI, 0, -1], [ PHI, 0,  1], [-PHI, 0, -1], [-PHI, 0,  1]
];

// Core Inverse Metaverse math engine
function computeInverseVectors({ seed, warp, density, tick }) {
  const xorFactor = ((seed ^ 0xffffff) & 0xff) / 255 || 0.5;
  const vectors = [];

  for (let layer = 1; layer <= density; layer++) {
    const ratio = layer / density;
    const scale = 90 + ratio * 220;
    const angle = tick + ratio * Math.PI * (xorFactor + 0.5);
    const hue = Number((((layer * Math.PI * 36) + tick * 60) % 360).toFixed(2));

    const projectedVertices = BASE_VERTICES.map(([x, y, z]) => {
      const cosY = Math.cos(angle), sinY = Math.sin(angle);
      const x1 = x * cosY - z * sinY;
      const z1 = z * cosY + x * sinY;

      const cosX = Math.cos(angle * 0.6), sinX = Math.sin(angle * 0.6);
      const y2 = y * cosX - z1 * sinX;
      const z2 = z1 * cosX + y * sinX;

      const invZ = 1 / Math.max(0.25, 3.2 - z2 * warp);
      return {
        x: Number((x1 * scale * invZ).toFixed(3)),
        y: Number((y2 * scale * invZ).toFixed(3)),
        depth: Number(z2.toFixed(3))
      };
    });

    vectors.push({ layer, hue, vertices: projectedVertices });
  }

  return vectors;
}

const server = new McpServer({
  name: "inverse-metaverse-connector",
  version: "1.0.0"
});

// Resource: Expose current pipeline state
server.resource(
  "pipeline-state",
  "metaverse://pipeline/state",
  async (uri) => ({
    contents: [
      {
        uri: uri.href,
        mimeType: "application/json",
        text: JSON.stringify(pipelineState, null, 2)
      }
    ]
  })
);

// Tool 1: Toggle or configure individual pipeline stages
server.tool(
  "set_pipeline_stage",
  "Set a stage (inputPool, computeLayer, linkingReasoning, executionLayer) to READY, ACTIVE, or IDLE.",
  {
    stage: z.enum(["inputPool", "computeLayer", "linkingReasoning", "executionLayer"]),
    status: z.enum(["READY", "ACTIVE", "IDLE"])
  },
  async ({ stage, status }) => {
    pipelineState[stage].status = status;
    return {
      content: [
        {
          type: "text",
          text: `Updated ${stage} to ${status}.\nCurrent State:\n${JSON.stringify(pipelineState, null, 2)}`
        }
      ]
    };
  }
);

// Tool 2: Execute the Inverse Metaverse vectorization pass
server.tool(
  "vectorize_inverse_metaverse",
  "Runs the 4-stage Inverse Metaverse calculation and returns projected 3D vectors and PI-mapped spectral hues.",
  {
    seed: z.number().int().default(0xffffff).describe("Hex/integer input seed for XOR white-point filter"),
    warp: z.number().min(-2).max(2).default(1.0).describe("Inverse perspective depth warp factor"),
    density: z.number().int().min(1).max(36).default(8).describe("Number of nested Platonic vector shells"),
    tick: z.number().default(0.5).describe("Phase rotation timestamp")
  },
  async ({ seed, warp, density, tick }) => {
    pipelineState.inputPool.seed = seed;
    pipelineState.computeLayer.status = "ACTIVE";
    pipelineState.computeLayer.warp = warp;
    pipelineState.computeLayer.density = density;
    pipelineState.linkingReasoning.status = "ACTIVE";
    pipelineState.executionLayer.status = "ACTIVE";

    const vectors = computeInverseVectors({ seed, warp, density, tick });
    pipelineState.executionLayer.lastVectorCount = vectors.length * BASE_VERTICES.length;

    return {
      content: [
        {
          type: "text",
          text: JSON.stringify(
            {
              pipeline: pipelineState,
              totalVerticesComputed: pipelineState.executionLayer.lastVectorCount,
              shells: vectors
            },
            null,
            2
          )
        }
      ]
    };
  }
);

// Start MCP server over stdio
async function main() {
  const transport = new StdioServerTransport();
  await server.connect(transport);
}

main().catch((err) => {
  console.error("MCP Server error:", err);
  process.exit(1);
});

import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { WebSocketServer } from "ws";
import { z } from "zod";

// Pipeline state matching your 4-stage control structure
const pipelineState = {
  stages: { 1: true, 2: true, 3: true, 4: true },
  labels: {
    inputPool: { status: "READY", expression: "&r == xora#FFFFFF", seed: 0xffffff },
    computeLayer: { status: "ACTIVE", model: "platonic_shapes(1, 1)", warp: 1.0, density: 16 },
    linkingReasoning: { status: "ACTIVE", mapping: "rainbow_tables(PI)", phase: Math.PI },
    executionLayer: { status: "ACTIVE", output: "pixels(r)" }
  }
};

// 1. Start WebSocket Bridge on port 8765
const wss = new WebSocketServer({ port: 8765 });

function broadcastState() {
  const payload = JSON.stringify({ type: "STATE_UPDATE", data: pipelineState });
  for (const client of wss.clients) {
    if (client.readyState === 1) {
      client.send(payload);
    }
  }
}

wss.on("connection", (ws) => {
  // Send current state immediately upon browser connection
  ws.send(JSON.stringify({ type: "STATE_UPDATE", data: pipelineState }));

  // Allow browser slider/button changes to sync back to the server
  ws.on("message", (raw) => {
    try {
      const msg = JSON.parse(raw.toString());
      if (msg.type === "CLIENT_SYNC" && msg.data) {
        Object.assign(pipelineState.stages, msg.data.stages || {});
        if (typeof msg.data.warp === "number") pipelineState.labels.computeLayer.warp = msg.data.warp;
        if (typeof msg.data.density === "number") pipelineState.labels.computeLayer.density = msg.data.density;
        broadcastState();
      }
    } catch {
      // Ignore malformed messages
    }
  });
});

            document.getElementById("warp").value = compute.warp;
            document.getElementById("density").value = compute.density;
          }
        }

      socket.onclose = () => {
        wsStatus.textContent = "WS Bridge: Offline (Retrying...)";
        wsStatus.style.color = "#f87171";
        setTimeout(connectBridge, 3000);

import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { WebSocketServer } from "ws";
import { z } from "zod";
import crypto from "node:crypto";

// Pipeline state matching your 4-stage control structure
const pipelineState = {
  stages: { 1: true, 2: true, 3: true, 4: true },
  labels: {
    inputPool: {
      status: "READY",
      expression: "&r == xora#FFFFFF",
      seed: 0xffffff,
      lastQuery: null
    },
    computeLayer: {
      status: "ACTIVE",
      model: "platonic_shapes(1, 1)",
      warp: 1.0,
      density: 16
    },
    linkingReasoning: {
      status: "ACTIVE",
      mapping: "rainbow_tables(PI)",
      phase: Math.PI
    },
    executionLayer: {
      status: "ACTIVE",
      output: "pixels(r)"
    }
  }
};

// 1. Start WebSocket Bridge on port 8765
const wss = new WebSocketServer({ port: 8765 });

function broadcastState() {
  const payload = JSON.stringify({ type: "STATE_UPDATE", data: pipelineState });
  for (const client of wss.clients) {
    if (client.readyState === 1) {
      client.send(payload);
    }
  }
}

wss.on("connection", (ws) => {
  ws.send(JSON.stringify({ type: "STATE_UPDATE", data: pipelineState }));

  ws.on("message", (raw) => {
    try {
      const msg = JSON.parse(raw.toString());
      if (msg.type === "CLIENT_SYNC" && msg.data) {
        Object.assign(pipelineState.stages, msg.data.stages || {});
        if (typeof msg.data.warp === "number") pipelineState.labels.computeLayer.warp = msg.data.warp;
        if (typeof msg.data.density === "number") pipelineState.labels.computeLayer.density = msg.data.density;
        broadcastState();
      }
    } catch {
      // Ignore malformed messages
    }
  });
});

// Helper: Convert scraped search text into a 24-bit hex seed and vector metrics
function hashResultsToVectorParams(query, results) {
  const combinedText = `${query}:${results.map((r) => r.title + r.url).join("|")}`;
  const hashHex = crypto.createHash("sha256").update(combinedText).digest("hex");

  // Extract 24-bit integer seed for &r == xora#FFFFFF
  const rawSeed = parseInt(hashHex.slice(0, 6), 16);
  const xorSeed = (rawSeed ^ 0xffffff) >>> 0;

  // Map result count to vector density (clamped between 6 and 36)
  const density = Math.min(36, Math.max(6, results.length * 4 || 12));

  // Map hash byte to inverse depth warp (-1.8 to 1.8)
  const warpByte = parseInt(hashHex.slice(6, 8), 16);
  const warp = Number((((warpByte / 255) * 3.6) - 1.8).toFixed(2));

  return { rawSeed, xorSeed, density, warp };
}

// Helper: Lightweight search scraper using DuckDuckGo Instant Answer API
async function scrapeSearchEngine(query, maxResults = 8) {
  const url = `https://duckduckgo.com{encodeURIComponent(query)}&format=json&no_html=1&skip_disambig=1`;
  const response = await fetch(url, {
    headers: { "User-Agent": "InverseMetaverseMCP/1.2" }
  });

  if (!response.ok) {
    throw new Error(`Search endpoint returned HTTP ${response.status}`);
  }

  const data = await response.json();
  const extracted = [];

  if (data.AbstractText && data.AbstractURL) {
    extracted.push({
      title: data.Heading || query,
      snippet: data.AbstractText,
      url: data.AbstractURL
    });
  }

  if (Array.isArray(data.RelatedTopics)) {
    for (const item of data.RelatedTopics) {
      if (extracted.length >= maxResults) break;
      if (item.Text && item.FirstURL) {
        extracted.push({
          title: item.Text.split(" - ")[0].slice(0, 80),
          snippet: item.Text,
          url: item.FirstURL
        });
      } else if (Array.isArray(item.Topics)) {
        for (const sub of item.Topics) {
          if (extracted.length >= maxResults) break;
          if (sub.Text && sub.FirstURL) {
            extracted.push({
              title: sub.Text.split(" - ")[0].slice(0, 80),
              snippet: sub.Text,
              url: sub.FirstURL
            });
          }
        }
      }
    }
  }

  return extracted;
}

// 2. Initialize MCP Server
const server = new McpServer({
  name: "inverse-metaverse-connector",
  version: "1.2.0"
});

const STAGE_MAP = {
  inputPool: 1,
  computeLayer: 2,
  linkingReasoning: 3,
  executionLayer: 4
};

// Tool 1: Toggle or configure individual pipeline stages
server.tool(
  "set_pipeline_stage",
  "Set a stage (inputPool, computeLayer, linkingReasoning, executionLayer) to READY, ACTIVE, or IDLE and stream to the browser.",
  {
    stage: z.enum(["inputPool", "computeLayer", "linkingReasoning", "executionLayer"]),
    status: z.enum(["READY", "ACTIVE", "IDLE"])
  },
  async ({ stage, status }) => {
    const stageId = STAGE_MAP[stage];
    pipelineState.stages[stageId] = status !== "IDLE";
    pipelineState.labels[stage].status = status;

    broadcastState();

    return {
      content: [
        {
          type: "text",
          text: `Streamed update to ${wss.clients.size} connected browser(s): ${stage} -> ${status}`
        }
      ]
    };
  }
);

// Tool 2: Manual vector parameter override
server.tool(
  "update_inverse_vectors",
  "Manually updates inverse depth warp, vector density, and seed in the live browser canvas over WebSockets.",
  {
    seed: z.number().int().optional().describe("Hex/integer input seed for XOR white-point filter"),
    warp: z.number().min(-2).max(2).optional().describe("Inverse perspective depth warp factor"),
    density: z.number().int().min(4).max(36).optional().describe("Number of nested Platonic vector shells")
  },
  async ({ seed, warp, density }) => {
    if (seed !== undefined) pipelineState.labels.inputPool.seed = seed;
    if (warp !== undefined) pipelineState.labels.computeLayer.warp = warp;
    if (density !== undefined) pipelineState.labels.computeLayer.density = density;

    broadcastState();

    return {
      content: [
        {
          type: "text",
          text: `Broadcasted vector parameters to ${wss.clients.size} browser(s):\n${JSON.stringify(pipelineState.labels, null, 2)}`
        }
      ]
    };
  }
);

// Tool 3: Web-Scraping AI Agent Tool
server.tool(
  "scrape_and_vectorize",
  "Scrapes live search engine results for a query, hashes them through the 01 Input Pool XOR filter, and streams updated geometry to the browser canvas.",
  {
    query: z.string().min(1).describe("Search topic or query to scrape and vectorize"),
    maxResults: z.number().int().min(1).max(15).default(8).describe("Maximum number of search results to ingest")
  },
  async ({ query, maxResults }) => {
    const results = await scrapeSearchEngine(query, maxResults);
    const { rawSeed, xorSeed, density, warp } = hashResultsToVectorParams(query, results);

    // Activate all 4 pipeline stages and apply derived parameters
    pipelineState.stages = { 1: true, 2: true, 3: true, 4: true };
    pipelineState.labels.inputPool.status = "READY";
    pipelineState.labels.inputPool.seed = xorSeed;
    pipelineState.labels.inputPool.lastQuery = query;

    pipelineState.labels.computeLayer.status = "ACTIVE";
    pipelineState.labels.computeLayer.density = density;
    pipelineState.labels.computeLayer.warp = warp;

    pipelineState.labels.linkingReasoning.status = "ACTIVE";
    pipelineState.labels.executionLayer.status = "ACTIVE";

    broadcastState();

    return {
      content: [
        {
          type: "text",
          text: JSON.stringify(
            {
              query,
              resultsScraped: results.length,
              inputPoolHash: `0x${rawSeed.toString(16).toUpperCase().padStart(6, "0")}`,
              xoraWhitePointSeed: `0x${xorSeed.toString(16).toUpperCase().padStart(6, "0")}`,
              computedParameters: { density, warp },
              activeBrowserClients: wss.clients.size,
              scrapedItems: results
            },
            null,
            2
          )
        }
      ]
    };
  }
);

// Start MCP transport
async function main() {
  const transport = new StdioServerTransport();
  await server.connect(transport);
}

main().catch((err) => {
  console.error("Server error:", err);
  process.exit(1);
});
